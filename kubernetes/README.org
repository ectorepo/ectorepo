#+begin_src emacs-lisp :var nrepos=100 :results value table :exports code :noweb yes :eval query
(setq -gql-data
      (ghub-graphql
       (graphql-query ((organization
                        :arguments ((login . "kubernetes"))
                        (repositories
                         :arguments ((first . 100)
                                     (orderBy . ((field . UPDATED_AT)
                                                 (direction . DESC))))
                         (edges
                          (node (owner login)
                                name
                                (defaultBranchRef prefix name)
                                url
                                updatedAt
                                diskUsage
                                isFork
                                isArchived))))))))

(thread-first
  (thread-last
    (a-get* (nthcdr 0 -gql-data) 'data 'organization 'repositories 'edges)
    (mapcar (lambda (el) (a-get* el 'node)))

    ;; filter archived repos
    (seq-filter (lambda (el) (not (a-get* el 'isFork))))
    (seq-filter (lambda (el) (not (a-get* el 'isArchived))))
    (mapcar (lambda (el)
              (flatten-list
               (list
                (a-get* el 'diskUsage)
                (a-get* el 'owner 'login)
                (a-get* el 'defaultBranchRef 'prefix)
                (a-get* el 'defaultBranchRef 'name)
                (mapcar (lambda (repo-keys)
                          (a-get* el repo-keys))
                        '(name
                          updatedAt
                          isFork
                          isArchived
                          url))))))))
#+end_src

Filter the results, generate XML

#+name: kubernetesReposXML
#+begin_src emacs-lisp :var gqldata=kubernetesRepos repos-exclude=kubernetesReposExclude :results value html
(setq -gql-data gqldata)

;; no repos-core variable
;; (repos-core (flatten-list repos- core))

(let* ((repos-exclude (flatten-list repos-exclude)))
  (thread-first
    (thread-last
      (a-get* (nthcdr 0 gqldata) 'data 'organization 'repositories 'edges)
      (mapcar (lambda (el) (a-get* el 'node)))

      ;; filter archived repos
      (seq-filter (lambda (el) (not (a-get* el 'isArchived))))

      ;; filter repos in reposExclude list
      (seq-filter (lambda (el) (not (member (a-get* el 'name) repos-exclude))))
      (mapcar (lambda (el)
                (let* ((raw-name (a-get* el 'name))

                       ;; (repo-core? (member raw-name repos-core))

                       (path-dirs (list "kubernetes" raw-name))

                       ;; (path-dirs (cond (repo-core? (list "core" raw-name))
                       ;;                 (t (list "misc" raw-name))))

                       (path (string-join path-dirs "/"))
                       (ref (concat (a-get* el 'defaultBranchRef 'prefix)
                                    (a-get* el 'defaultBranchRef 'name)))
                       (name (string-join (list (a-get* el 'owner 'login)
                                                (a-get* el 'name)) "/")))
                  (concat "<project"
                          " name=\"" name
                          "\" path=\"" path
                          "\" revision=\"" ref "\" remote=\"github\"/>")))))
    (cl-sort 'string-lessp :key 'downcase)
    (string-join "\n")))
#+end_src
