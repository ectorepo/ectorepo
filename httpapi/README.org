#+begin_src emacs-lisp :var login=apis-guruLogin nrepos=(identity 100) :results vector value :exports code
(let* ((q '(query
            [($login String!)
             ($first Int!)
             ($orderBy RepositoryOrder!)]
            (organization
             [(login $login)]
             (repositories
              [(first $first)
               (orderBy $orderBy)]
              (edges
               (node
                (owner login)
                name
                (defaultBranchRef prefix name)
                url
                updatedAt
                diskUsage
                isFork
                isArchived))))))
       (vars `((login . ,login)
               (first . ,nrepos)
               (orderBy . ((field . "UPDATED_AT")
                           (direction . "DESC")))))
       (response (cdr (ghub-query q vars
                   :paginate 100
                   :synchronous t))))
  ;; (setq gql-response response)
  (let-alist response .organization.repositories))
#+end_src

** APIs.guru Repos XML

Filter the results, generate XML

#+name: apis-guruReposXML
#+begin_src emacs-lisp :var repos=apis-guruRepos repos-exclude=apis-guruReposExclude login=apis-guruLogin :results value html
(let* ((repos-exclude (flatten-list repos-exclude)))
  (thread-first
    (thread-last
      repos
      (seq-filter (lambda (el) (not (a-get* el 'isArchived))))
      (seq-filter (lambda (el) (not (a-get* el 'isFork))))
      (seq-filter (lambda (el) (not (member (a-get* el 'name) repos-exclude))))
      (mapcar (lambda (el)
                (let* ((raw-name (a-get* el 'name))
                       (path-dirs (list login raw-name))

                       ;; (path-dirs (cond (repo-core? (list "core" raw-name))
                       ;;                 (t (list "misc" raw-name))))

                       (path (string-join path-dirs "/"))
                       (ref (concat (a-get* el 'defaultBranchRef 'prefix)
                                    (a-get* el 'defaultBranchRef 'name)))
                       (name (string-join (list (a-get* el 'owner 'login)
                                                (a-get* el 'name)) "/")))
                  (concat "<project"
                          " name=\"" name
                          "\" path=\"" path
                          "\" revision=\"" ref "\" remote=\"github\"/>")))))
    (cl-sort 'string-lessp :key 'downcase)
    (string-join "\n")))
#+end_src
